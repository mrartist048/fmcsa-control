// Perfected Delete User Functionality across all database nodes and tabs async function deleteUserRecord(userName) { if (!confirm(`Are you sure you want to delete user "${userName}"? This will permanently erase their session, call logs, and shift reports from all tabs.`)) { return; } try { // 1. Fetch current database snapshot to find matching nodes safely let [sRes, cRes, rRes] = await Promise.all([ fetch(`${FIREBASE_DB_URL}sessions/${currentClient}.json`), fetch(`${FIREBASE_DB_URL}call_logs/${currentClient}.json`), fetch(`${FIREBASE_DB_URL}shift_reports/${currentClient}.json`) ]); let sessionsData = await sRes.json() || {}; let callLogsData = await cRes.json() || {}; let shiftReportsData = await rRes.json() || {}; let deletePromises = []; // Helper to check and push delete request if user matches function checkAndDelete(nodeName, dataObj) { Object.keys(dataObj).forEach(key => { let item = dataObj[key]; let candidateName = (typeof item === 'object' && item !== null) ? (item.nickname || item.name || key) : key; // Match by exact name, key, or cleaned key if (key.toLowerCase() === userName.toLowerCase() || candidateName.toLowerCase() === userName.toLowerCase()) { deletePromises.push(fetch(`${FIREBASE_DB_URL}${nodeName}/${currentClient}/${key}.json`, { method: 'DELETE' })); } }); } checkAndDelete('sessions', sessionsData); checkAndDelete('call_logs', callLogsData); checkAndDelete('shift_reports', shiftReportsData); // Also direct standard keys just in case let safeKey = userName.replace(/[.#$\/\[\]]/g, "_"); deletePromises.push( fetch(`${FIREBASE_DB_URL}sessions/${currentClient}/${userName}.json`, { method: 'DELETE' }), fetch(`${FIREBASE_DB_URL}sessions/${currentClient}/${safeKey}.json`, { method: 'DELETE' }), fetch(`${FIREBASE_DB_URL}call_logs/${currentClient}/${userName}.json`, { method: 'DELETE' }), fetch(`${FIREBASE_DB_URL}call_logs/${currentClient}/${safeKey}.json`, { method: 'DELETE' }), fetch(`${FIREBASE_DB_URL}shift_reports/${currentClient}/${userName}.json`, { method: 'DELETE' }), fetch(`${FIREBASE_DB_URL}shift_reports/${currentClient}/${safeKey}.json`, { method: 'DELETE' }) ); await Promise.all(deletePromises); // Clear from local cache immediately if (cachedSessions) delete cachedSessions[userName]; if (cachedCallLogs) delete cachedCallLogs[userName]; if (cachedReports) delete cachedReports[userName]; alert(`User "${userName}" and all their calling/shift report records have been permanently deleted.`); fetchAdminData(); } catch (err) { console.error("Delete error:", err); alert("Failed to delete user completely. Check console for details."); } }
