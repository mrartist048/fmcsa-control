// Delete User Functionality across all database nodes async function deleteUserRecord(userName) { if (!confirm(`Are you sure you want to delete user "${userName}"? This will remove their session, call logs, and shift reports permanently.`)) { return; } try { let safeKey = userName.replace(/[.#$\/\[\]]/g, "_"); // Fetch all data to find any keys matching the user name case-insensitively or exact await Promise.all([ fetch(`${FIREBASE_DB_URL}sessions/${currentClient}/${userName}.json`, { method: 'DELETE' }), fetch(`${FIREBASE_DB_URL}sessions/${currentClient}/${safeKey}.json`, { method: 'DELETE' }), fetch(`${FIREBASE_DB_URL}call_logs/${currentClient}/${userName}.json`, { method: 'DELETE' }), fetch(`${FIREBASE_DB_URL}call_logs/${currentClient}/${safeKey}.json`, { method: 'DELETE' }), fetch(`${FIREBASE_DB_URL}shift_reports/${currentClient}/${userName}.json`, { method: 'DELETE' }), fetch(`${FIREBASE_DB_URL}shift_reports/${currentClient}/${safeKey}.json`, { method: 'DELETE' }) ]); alert(`User "${userName}" successfully deleted from all records.`); fetchAdminData(); } catch (err) { console.error("Delete error:", err); alert("Failed to delete user. Check console for details."); } }
